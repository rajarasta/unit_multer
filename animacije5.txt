import React, { useState, useMemo } from 'react';

// Broj stupaca i redaka za mrežu animacije
const COLS = 10;
const ROWS = 18;
const NUM_TILES = COLS * ROWS;
// Trajanje animacije u milisekundama - POVEĆANO NA 4s
const ANIMATION_DURATION = 4000;

// Definicija keyframe animacije - sada koristi i --direction varijablu
const waveKeyframe = `
  @keyframes wave {
    0%, 100% {
      transform: translate3d(0px, 0px, 0px) scale(1) rotateX(0) rotateY(0);
      box-shadow: 0 1px 2px rgba(0,0,0,0.15);
      opacity: 1;
    }
    50% {
      /* Množi pomak po X osi sa smjerom */
      transform: translate3d(calc(var(--tx, 0px) * var(--direction, 1)), var(--ty, 0px), var(--tz, 0px)) scale(var(--s, 0.85)) rotateX(var(--rx, 0deg)) rotateY(var(--ry, 0deg));
      box-shadow: 0 calc(var(--thick, 8px) * 1.6) calc(var(--thick, 1px) * 2.4) rgba(0,0,0,0.28);
      opacity: 0.95;
    }
  }
`;

// Glavna komponenta aplikacije
export default function App() {
  const [theme, setTheme] = useState('day');
  const [isAnimating, setIsAnimating] = useState(false);
  // NOVO: Stanje za smjer animacije (1 = desno, -1 = lijevo)
  const [animationDirection, setAnimationDirection] = useState(1);

  // Stanje za tekst koji se prikazuje
  const [displayText, setDisplayText] = useState('Svijetlo. Toplina. Energija dana nosi sve pred sobom.');

  // Podaci za teme
  const themes = {
    day: {
      outerBgClass: 'bg-sky-200',
      innerBg: 'from-orange-300 to-sky-400',
      textColor: 'text-slate-800',
      text: 'Svijetlo. Toplina. Energija dana nosi sve pred sobom.',
    },
    night: {
      outerBgClass: 'bg-slate-900',
      innerBg: 'from-indigo-800 to-purple-900',
      textColor: 'text-white',
      text: 'Tišina. Mir. Zvijezde ispisuju priče na tamnom platnu.',
    },
  };

  // Deterministički PRNG za stabilne vrijednosti
  const { animationDelays, ampX, ampY, ampZ, thickness, scales, rotX, rotY } = useMemo(() => {
    let seed = 1337;
    function rand() {
      seed = (seed * 16807) % 2147483647;
      return (seed % 1000) / 1000;
    }

    const delays = Array.from({ length: NUM_TILES }, () => Math.round(rand() * (ANIMATION_DURATION / 2)));
    const aY = Array.from({ length: NUM_TILES }, () => Math.round(10 + rand() * 30));
    const aX = Array.from({ length: NUM_TILES }, () => Math.round(80 + rand() * 100));
    const aZ = Array.from({ length: NUM_TILES }, () => Math.round(rand() * 40));
    const thick = Array.from({ length: NUM_TILES }, () => Math.round(4 + rand() * 10));
    const s = Array.from({ length: NUM_TILES }, () => 0.75 + rand() * 0.2);
    const rX = Array.from({ length: NUM_TILES }, () => Math.round(-20 + rand() * 40));
    const rY = Array.from({ length: NUM_TILES }, () => Math.round(-20 + rand() * 40));

    // ISPRAVAK: Vraćamo objekt s ispravnim imenima varijabli
    return { 
      animationDelays: delays, 
      ampX: aX, 
      ampY: aY, 
      ampZ: aZ, 
      thickness: thick, 
      scales: s, 
      rotX: rX, 
      rotY: rY 
    };
  }, []);

  // Funkcija koja se poziva na klik
  const handleThemeChange = () => {
    if (isAnimating) return;

    const nextTheme = theme === 'day' ? 'night' : 'day';

    // Postavi smjer ovisno o sljedećoj temi
    setAnimationDirection(nextTheme === 'night' ? 1 : -1);
    setIsAnimating(true);

    setTimeout(() => {
        setTheme(nextTheme);
    }, 300);

    const maxDelay = Math.max(...animationDelays);
    setTimeout(() => {
      setIsAnimating(false);
      setDisplayText(themes[nextTheme].text);
    }, ANIMATION_DURATION + maxDelay + 80);
  };

  const currentTheme = themes[theme];

  return (
    <>
      <style>{waveKeyframe}</style>
      <div
        className={`
          w-full h-screen 
          flex items-center justify-center 
          font-sans 
          overflow-hidden
          ${currentTheme.outerBgClass}
        `}
        style={{ transition: 'none' }}
      >
        <div
          onClick={handleThemeChange}
          className="relative w-11/12 max-w-md cursor-pointer"
          style={{ 
            aspectRatio: `${COLS / (ROWS * 1.5)}`,
            perspective: '1000px' 
          }}
        >
          {!isAnimating && (
            <p
              className={`
                absolute inset-0 flex items-center justify-center
                p-8 md:p-12
                text-xl md:text-2xl 
                font-light text-center leading-relaxed
                z-10
                ${currentTheme.textColor}
              `}
            >
              {displayText}
            </p>
          )}

          <div 
            className="grid w-full h-full shadow-2xl"
            style={{ 
                transformStyle: 'preserve-3d', 
                gridTemplateColumns: `repeat(${COLS}, 1fr)`, 
                gridTemplateRows: `repeat(${ROWS}, 1fr)`, 
                display: 'grid', 
            }}
          >
            {Array.from({ length: NUM_TILES }).map((_, i) => {
              const col = i % COLS;
              const row = Math.floor(i / COLS);
              const { delay, tx, ty, tz, thick, scale, rx, ry } = {
                delay: animationDelays[i],
                tx: ampX[i],
                ty: ampY[i],
                tz: ampZ[i],
                thick: thickness[i],
                scale: scales[i],
                rx: rotX[i],
                ry: rotY[i],
              };

              return (
                <div
                  key={i}
                  // ISPRAVAK: Primjena gradijenta putem Tailwind klasa
                  className={`w-full h-full transition-all duration-1000 bg-gradient-to-b ${currentTheme.innerBg}`}
                  style={{
                    backgroundSize: `${COLS * 100}% ${ROWS * 100}%`,
                    backgroundPosition: `${(100 / (COLS - 1)) * col}% ${(100 / (ROWS - 1)) * row}%`,
                    // ISPRAVAK: Standardna sintaksa za CSS varijable
                    '--direction': animationDirection,
                    '--tx': `${tx}px`,
                    '--ty': `${-ty}px`,
                    '--tz': `${tz}px`,
                    '--thick': `${thick}px`,
                    '--s': scale,
                    '--rx': `${rx}deg`,
                    '--ry': `${ry}deg`,
                    border: '1px solid rgba(255,255,255,0.04)',
                    boxShadow: `0 ${Math.max(2, Math.floor(thick/2))}px ${Math.floor(thick*1.4)}px rgba(0,0,0,0.12)`,
                    borderRadius: 0,
                    animation: isAnimating ? `wave ${ANIMATION_DURATION}ms ${delay}ms cubic-bezier(0.34, 1.56, 0.64, 1)` : 'none',
                    transformStyle: 'preserve-3d',
                  }}
                />
              );
            })}
          </div>

          {isAnimating && (
            <div style={{ position: 'absolute', left: 0, right: 0, bottom: 12, display: 'flex', justifyContent: 'center', pointerEvents: 'none' }}>
              <div style={{ width: '85%', maxWidth: 980, padding: '12px 16px', borderRadius: 10, backdropFilter: 'blur(6px)', background: 'rgba(255,255,255,0.02)', color: currentTheme.textColor, textAlign: 'center' }}>
                {displayText}
              </div>
            </div>
          )}
        </div>
      </div>
    </>
  );
}
