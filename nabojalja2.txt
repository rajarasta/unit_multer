import React, { useState, useMemo } from 'react';

// Broj stupaca i redaka za mrežu animacije - VRAĆENO NA STARO
const COLS = 10;
const ROWS = 18;
const NUM_TILES = COLS * ROWS;
// Trajanje animacije u milisekundama
const ANIMATION_DURATION = 1500;

// Definicija keyframe animacije - Sjena se pojavljuje i nestaje
const waveKeyframe = `
  @keyframes wave {
    0%, 100% {
      transform: scale(1) rotateX(0) rotateY(0);
      box-shadow: none;
      border-color: rgba(0,0,0,0); /* Rub je proziran na početku i kraju */
      opacity: 1;
    }
    50% {
      transform: scale(var(--s, 0.85)) rotateX(var(--rx, 0deg)) rotateY(var(--ry, 0deg)) rotateZ(var(--rz, 0deg));
      box-shadow: inset 0 2px 2px rgba(255,255,255,0.2), inset 0 -2px 2px rgba(0,0,0,0.2), 0 25px 35px rgba(0,0,0,0.4);
      border-color: rgba(0,0,0,0.1); /* Rub je vidljiv u sredini animacije */
      opacity: 0.95;
    }
  }
`;

// Glavna komponenta aplikacije
export default function App() {
  const [theme, setTheme] = useState('day');
  const [isAnimating, setIsAnimating] = useState(false);
  const [displayText, setDisplayText] = useState('Svijetlo. Toplina. Energija dana nosi sve pred sobom.');

  const themes = {
    day: {
      outerBgClass: 'bg-sky-200',
      innerBg: 'from-orange-300 to-sky-400',
      textColor: 'text-slate-800',
      text: 'Svijetlo. Toplina. Energija dana nosi sve pred sobom.',
    },
    night: {
      outerBgClass: 'bg-slate-900',
      innerBg: 'from-indigo-800 to-purple-900',
      textColor: 'text-white',
      text: 'Tišina. Mir. Zvijezde ispisuju priče na tamnom platnu.',
    },
  };

  // Memoizirane vrijednosti za animaciju
  const { animationDelays, thickness, scales, rotX, rotY, rotZ } = useMemo(() => {
    let seed = 1337;
    function rand() {
      seed = (seed * 16807) % 214783647;
      return (seed % 1000) / 1000;
    }

    const delays = Array.from({ length: NUM_TILES }, () => Math.round(rand() * (ANIMATION_DURATION / 2)));
    const thick = Array.from({ length: NUM_TILES }, () => Math.round(4 + rand() * 10));
    const s = Array.from({ length: NUM_TILES }, () => 0.6 + rand() * 0.4);
    const rX = Array.from({ length: NUM_TILES }, () => Math.round(-70 + rand() * 140));
    const rY = Array.from({ length: NUM_TILES }, () => Math.round(-70 + rand() * 140));
    const rZ = Array.from({ length: NUM_TILES }, () => Math.round(-20 + rand() * 40));

    return { 
        animationDelays: delays, 
        thickness: thick, 
        scales: s, 
        rotX: rX, 
        rotY: rY,
        rotZ: rZ
    };
  }, []);
  
  // Kreiranje mreže s karakterima teksta
  const textGrid = useMemo(() => {
    const grid = Array(NUM_TILES).fill('');
    
    // PROMJENA: Tekst se sada automatski prelama kao odlomak
    const fullText = "Svijetlo. Toplina. Energija dana nosi sve pred sobom.";
    const words = fullText.split(' ');
    let currentLine = '';
    const lines = [];

    words.forEach(word => {
        if ((currentLine + ' ' + word).trim().length > COLS) {
            lines.push(currentLine);
            currentLine = word;
        } else {
            currentLine = (currentLine + ' ' + word).trim();
        }
    });
    lines.push(currentLine); // Dodaj zadnji red

    const startRow = Math.floor(ROWS / 2) - Math.floor(lines.length / 2);

    lines.forEach((line, lineIndex) => {
        const currentRow = startRow + lineIndex;
        // Svaki red se i dalje centrira za najbolji izgled
        const startCol = Math.floor(COLS / 2) - Math.floor(line.length / 2);
        line.split('').forEach((char, charIndex) => {
            const currentCol = startCol + charIndex;
            if (currentCol >= 0 && currentCol < COLS) {
                const index = currentRow * COLS + currentCol;
                if (index < NUM_TILES) {
                    grid[index] = char;
                }
            }
        });
    });
    return grid;
  }, []);


  const handleThemeChange = () => {
    if (isAnimating) return;
    
    setIsAnimating(true);
    
    const maxDelay = Math.max(...animationDelays);
    setTimeout(() => {
      setIsAnimating(false);
    }, ANIMATION_DURATION + maxDelay + 80);
  };

  const currentTheme = themes[theme];

  return (
    <>
      <style>{waveKeyframe}</style>
      <div
        className={`w-full h-screen flex items-center justify-center font-sans overflow-hidden ${currentTheme.outerBgClass}`}
        style={{ transition: 'none' }}
      >
        <div
          onClick={handleThemeChange}
          className="relative w-11/12 max-w-md cursor-pointer"
          style={{ aspectRatio: `${COLS / (ROWS * 1.5)}`, perspective: '800px' }}
        >
          {/* UKLONJEN STATIČNI TEKST ODAVDE */}
          <div 
            className="grid w-full h-full"
            style={{ 
                transformStyle: 'preserve-3d', 
                gridTemplateColumns: `repeat(${COLS}, 1fr)`, 
                gridTemplateRows: `repeat(${ROWS}, 1fr)`, 
                display: 'grid', 
            }}
          >
            {Array.from({ length: NUM_TILES }).map((_, i) => {
              const col = i % COLS;
              const row = Math.floor(i / COLS);
              const { delay, thick, scale, rx, ry, rz } = {
                delay: animationDelays[i],
                thick: thickness[i], scale: scales[i], rx: rotX[i], ry: rotY[i], rz: rotZ[i],
              };
              
              // Dohvati karakter za ovu pločicu
              const char = textGrid[i];

              return (
                <div
                  key={i}
                  // Dodane flex klase za centriranje teksta
                  className={`w-full h-full flex items-center justify-center transition-all duration-1000 bg-gradient-to-b ${currentTheme.innerBg}`}
                  style={{
                    backgroundSize: `${COLS * 100}% ${ROWS * 100}%`,
                    backgroundPosition: `${(100 / (COLS - 1)) * col}% ${(100 / (ROWS - 1)) * row}%`,
                    '--thick': `${thick}px`, '--s': scale, '--rx': `${rx}deg`, '--ry': `${ry}deg`, '--rz': `${rz}deg`,
                    border: '1px solid rgba(0,0,0,0)',
                    boxShadow: 'none',
                    borderRadius: 0,
                    animation: isAnimating ? `wave ${ANIMATION_DURATION}ms ${delay}ms cubic-bezier(0.34, 1.56, 0.64, 1)` : 'none',
                    transformStyle: 'preserve-3d',
                  }}
                >
                  {/* Span s karakterom unutar pločice. Font je ostao text-base. */}
                  <span className={`text-base font-light transition-colors duration-500 ${currentTheme.textColor}`} style={{transform: 'translateZ(2px)'}}>
                    {char}
                  </span>
                </div>
              );
            })}
          </div>
          {/* UKLONJEN STATIČNI TEKST ODAVDE */}
        </div>
      </div>
    </>
  );
}
