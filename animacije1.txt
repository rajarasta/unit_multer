import React, { useState, useMemo } from 'react';

// Broj stupaca i redaka za mrežu animacije
const COLS = 10;
const ROWS = 9;
const NUM_TILES = COLS * ROWS;
// Trajanje animacije u milisekundama
const ANIMATION_DURATION = 2000;

// Definicija keyframe animacije s 3D transformacijama
const waveKeyframe = `
  @keyframes wave {
    0%, 100% {
      transform: translateZ(0) rotateX(0);
      box-shadow: 0 2px 5px rgba(0,0,0,0.1);
      opacity: 1;
    }
    50% {
      transform: translateZ(30px) rotateX(-20deg);
      box-shadow: 0 35px 30px rgba(0,0,0,0.2);
      opacity: 0.9;
    }
  }
`;

// Glavna komponenta aplikacije
export default function App() {
  const [theme, setTheme] = useState('day');
  const [isAnimating, setIsAnimating] = useState(false);
  
  // Stanje za tekst koji se prikazuje
  const [displayText, setDisplayText] = useState('Svijetlo. Toplina. Energija dana nosi sve pred sobom.');
  const [textOpacity, setTextOpacity] = useState(1);

  // Podaci za teme
  const themes = {
    day: {
      innerBg: 'from-orange-300 to-sky-400',
      textColor: 'text-white',
      text: 'Svijetlo. Toplina. Energija dana nosi sve pred sobom.',
    },
    night: {
      innerBg: 'from-indigo-800 to-purple-900',
      textColor: 'text-white',
      text: 'Tišina. Mir. Zvijezde ispisuju priče na tamnom platnu.',
    },
  };

  // Generiramo nasumična kašnjenja animacije za svaku pločicu.
  const animationDelays = useMemo(
    () => Array.from({ length: NUM_TILES }, () => Math.random() * (ANIMATION_DURATION / 2)),
    []
  );

  // Funkcija koja se poziva na klik
  const handleThemeChange = () => {
    if (isAnimating) return; // Sprječava ponovno pokretanje animacije

    setIsAnimating(true);
    setTextOpacity(0); // Sakrij stari tekst

    const nextTheme = theme === 'day' ? 'night' : 'day';

    // Nakon kratkog vremena za fade-out teksta, mijenjamo temu
    setTimeout(() => {
        setTheme(nextTheme);
    }, 300);


    // Postavljamo timeout za resetiranje stanja nakon završetka animacije
    setTimeout(() => {
      setIsAnimating(false);
      setDisplayText(themes[nextTheme].text); // Postavi novi tekst
      setTextOpacity(1); // Prikaži novi tekst
    }, ANIMATION_DURATION);
  };

  const currentTheme = themes[theme];

  return (
    <>
      {/* Umetanje keyframe animacije direktno u DOM */}
      <style>{waveKeyframe}</style>
      
      {/* Vanjska komponenta (pozornica) - fiksna boja pozadine */}
      <div
        className={`
          w-full h-screen 
          flex items-center justify-center 
          font-sans 
          overflow-hidden
          bg-slate-900
        `}
      >
        {/* Unutarnja komponenta - dodana perspektiva za 3D efekt */}
        <div
          onClick={handleThemeChange}
          className="relative w-11/12 max-w-lg cursor-pointer"
          style={{ 
            aspectRatio: `${COLS / ROWS}`,
            perspective: '1000px' 
          }}
        >
          {/* Tekstualni sadržaj - Apsolutno pozicioniran iznad mreže */}
          <p
            className={`
              absolute inset-0 flex items-center justify-center
              p-6 md:p-8
              text-lg md:text-xl 
              font-light text-center leading-relaxed
              transition-opacity duration-500 ease-in-out
              z-10
              ${currentTheme.textColor}
            `}
            style={{ 
              opacity: textOpacity,
              textShadow: '0 2px 10px rgba(0,0,0,0.5)'
            }}
          >
            {displayText}
          </p>

          {/* Mreža animiranih pločica - dodan transform-style */}
          <div 
            className="grid grid-cols-10 grid-rows-9 w-full h-full rounded-2xl overflow-hidden shadow-2xl"
            style={{ transformStyle: 'preserve-3d' }}
          >
            {Array.from({ length: NUM_TILES }).map((_, i) => {
              const col = i % COLS;
              const row = Math.floor(i / COLS);
              return (
                <div
                  key={i}
                  className={`w-full h-full bg-gradient-to-b ${currentTheme.innerBg} transition-all duration-1000`}
                  style={{
                    // Svaka pločica prikazuje samo dio cijelog gradijenta
                    backgroundSize: `${COLS * 100}% ${ROWS * 100}%`,
                    backgroundPosition: `${(100 / (COLS - 1)) * col}% ${(100 / (ROWS - 1)) * row}%`,
                    // Primjena animacije ako je aktivna
                    animation: isAnimating
                      ? `wave ${ANIMATION_DURATION}ms ${animationDelays[i]}ms ease-in-out`
                      : 'none',
                  }}
                />
              );
            })}
          </div>
        </div>
      </div>
    </>
  );
}
